// Shared types — frontend ↔ backend wire format and UI domain model.
// The Python server (`src/poietes/server.py`) is the source of truth for
// the API shape; keep this file in sync if the server contract changes.
 
export type Role = "user" | "assistant";
export type Theme = "light" | "dark";
/**
 * Single-letter code identifying which philosopher's chunks dominated a
 * given retrieval. P=Plato, A=Aristotle, H=Hegel, K=Kant. The mapping
 * from corpus name → code lives server-side in `_CORPUS_TO_DOMINANT`
 * (src/poietes/server.py); the client never computes it.
 */
export type Dominant = "P" | "A" | "H" | "K";
// Only "oxblood" is wired up; the bronze/lapis prototype variants
// were never selected in App.tsx and have been retired from
// data.ts. Keep the type narrow so an accidental reintroduction is
// a compile error.
export type AccentKey = "oxblood";
 
/** A turn the LLM should see, used in `history` of /api/chat. */
export interface ChatHistoryMessage {
  role: Role;
  content: string;
}
 
/** /api/chat request body. */
export interface ChatRequest {
  query: string;
  history?: ChatHistoryMessage[];
  /** Per-request override; falls back to the server's startup `--speaker`. */
  speaker?: string | null;
  /** Ask the LLM to rewrite the follow-up into a standalone retrieval query. */
  rewrite?: boolean;
  /** Pinned chunk IDs the user has anchored from prior turns. */
  pinned_chunk_ids?: string[];
  /** Existing conversation to append this turn to. Omit to create a new one. */
  conversation_id?: string;
}
 
/** A retrieval result surfaced as a citation in the UI. */
export interface Citation {
  n: number;
  src: string;
  text: string;
  chunk_id?: string;
  score?: number;
  /** Character who spoke this passage in the source dialogue, when available. */
  speaker?: string | null;
  /** True when this citation is anchored from a prior turn rather than freshly retrieved. */
  pinned?: boolean;
}
 
/** /api/chat response body. */
export interface ChatResponse {
  response: string;
  paragraphs: string[];
  citations: Citation[];
  dominant: Dominant;
  abstained: boolean;
  backend: string;
  /** ID of the conversation this turn was appended to (always present). */
  conversation_id: string;
  /** Sidebar title — only present on the first turn (creation) of a conversation. */
  title?: string;
  /** Pinned chunk IDs the client sent that the server couldn't resolve —
   *  typically stale references after a re-ingest. Client should drop
   *  them from local pin state. Absent when nothing was dropped. */
  missing_pinned_ids?: string[];
}

/** Output of `splitCitations` — text fragments interleaved with cite markers. */
export interface CiteFragment {
  kind: "cite";
  n: number;
}
export type CitationFragment = string | CiteFragment;
 
export function isCiteFragment(x: CitationFragment): x is CiteFragment {
  // `CitationFragment = string | CiteFragment`, so `typeof x === "object"`
  // already narrows to CiteFragment — checking `kind === "cite"` is
  // tautological under that union.
  return typeof x === "object";
}

/** A single work surfaced by GET /api/works. */
export interface Work {
  corpus: string;
  author: string;
  /** Canonical identifier — what the on-disk chunks carry and what the
   *  server expects in subsequent /api/works/<work> calls. For Greek
   *  this is Perseus beta-code (e.g., `*ti/maios`); use `display_title`
   *  for labels. */
  work: string;
  /** Render-ready title. Polytonic Unicode for Greek, identical to
   *  `work` for other languages. */
  display_title: string;
  lang: string;
  chunks: number;
}
 
/** One source-language chunk in the reader column. */
export interface SourceChunk {
  chunk_id: string;
  section: string | null;
  book: string | null;
  page: string | null;
  speaker: string | null;
  text: string;
  ord: number;
}
 
/** GET /api/works/:work?lang=... response. */
export interface WorkRange {
  /** Integer surrogate from the `works` table. Used to address
   *  work-id-keyed endpoints (e.g. /api/works/:work_id/occurrence_map)
   *  without an extra round trip through /api/works. */
  work_id: number;
  corpus: string;
  author: string;
  work: string;
  /** Polytonic Unicode for Greek beta-code titles; otherwise equals `work`. */
  display_title: string;
  lang: string;
  total: number;
  offset: number;
  limit: number;
  chunks: SourceChunk[];
}
 
/** One term-occurrence span in a chunk, as returned by the occurrence
 *  map / by-chunk endpoints. The pair (source_start, source_end) is a
 *  character offset into the chunk's `text` field as the reader
 *  receives it. */
export interface TermOccurrenceSpan {
  /** term_occurrence row id (not term id). */
  id: number;
  term_id: number;
  source_start: number;
  source_end: number;
  /** Surface form actually present in the chunk (e.g. "ὁμωνύμως"). */
  source_form: string;
  /** Canonical lemma from the `terms` table (e.g. "ὁμώνυμα"). */
  lemma: string;
  transliteration: string | null;
  short_gloss: string;
}
 
/** GET /api/works/:work_id/occurrence_map response. */
export interface OccurrenceMap {
  work_id: number;
  lang: string;
  from: number;
  limit: number;
  /** Keyed by chunk_id, in document order. Chunks without any
   *  occurrence are omitted (so the map can stay small). */
  by_chunk: Record<string, TermOccurrenceSpan[]>;
}

/** One sample site returned by /api/words/lookup. */
export interface WordSample {
  chunk_id: string;
  section: string | null;
  surface: string;
  surface_start: number;
  surface_end: number;
  /** Whitespace-normalised window around the surface. */
  snippet: string;
  /** Offsets within `snippet` where the matched surface lives.
   *  Let the UI wrap `snippet.slice(match_start, match_end)` in a
   *  highlight span without a string-search heuristic that would
   *  break on lemmas appearing multiple times in the same window. */
  match_start: number;
  match_end: number;
}
 
/** One highlight span returned by /api/works/:id/lemma_highlights. */
export interface LemmaHighlightSpan {
  start: number;
  end: number;
}
 
/** GET /api/works/:id/lemma_highlights response. */
export interface LemmaHighlights {
  work_id: number;
  lang: string;
  lemma: string;
  from: number;
  limit: number;
  /** Keyed by chunk_id. A chunk that doesn't contain the lemma at
   *  all is omitted, so a frontend `.[chunkId] ?? []` lookup is
   *  cheap. Spans within a chunk are sorted by `start`. */
  by_chunk: Record<string, LemmaHighlightSpan[]>;
}
 
/** GET /api/words/lookup response. */
export interface WordLookup {
  surface: string;
  lemma: string;
  lang: string;
  edition_id: number;
  work_id: number;
  work_title: string;
  occurrence_count: number;
  /** Cursor where this page of samples started (echoes the request). */
  samples_offset: number;
  /** Cursor for the next page, or null when there's nothing further.
   *  The rail's "Show more" button is disabled while this is null. */
  next_samples_offset: number | null;
  /** Universal Dependencies coarse part-of-speech (NOUN, VERB, ADJ, …).
   *  Null for legacy chunk_lemmas rows seeded before migration 0008. */
  pos: string | null;
  /** Morph features pulled from the UD feats string for direct
   *  rendering. `Masc | Fem | Neut | Com` for gender, `Sing | Plur |
   *  Dual` for number. Null when the lemmatizer didn't tag the
   *  feature or when the chunk_lemmas row predates 0008. */
  gender: string | null;
  number: string | null;
  /** Raw UD morph string (`Key=Val|Key=Val|...`). Provided for
   *  forward compatibility — clients can pull case / tense / etc.
   *  without an endpoint change. */
  feats: string | null;
  samples: WordSample[];
}

// ────────────────────────────────────────────────────────────────
// Cross-references
// ────────────────────────────────────────────────────────────────
//
// Shared between the API client (chat.ts) and the ReferenceCard
// popover (lib/ReferenceCard.tsx). The two used to maintain two
// structurally-identical copies plus a `toCardReference` shim in
// Reader/index.tsx; lifting the types here lets them share one
// definition without circular import.
 
/** Target-side metadata on a cross-reference row. */
export interface ReferenceTarget {
  author: string;
  work: string;
  /** Non-null when the target work is in our local corpus index. */
  in_corpus_work_id: number | null;
  /** Non-null when the target is a specific in-corpus chunk. */
  chunk_id: string | null;
  /** Canonical citation string ("VIII.6", "Republic 514a"). */
  locator: string | null;
  locator_payload: Record<string, unknown> | null;
  sort_key: string | null;
}
 
/** One cross-reference row from GET /api/chunks/:id/references and
 *  related endpoints. */
export type ReferenceType =
  | "cites"
  | "quotes"
  | "paraphrases"
  | "alludes"
  | "comments_on"
  | "translates"
  | "disputes"
  | "corrects";
 
/** One cross-reference row — the unified shape used by both the API
 *  client and the popover component. Replaces the older
 *  `ReferenceRow` (chat.ts) and `Reference` (ReferenceCard.tsx)
 *  duplicates. */
export interface Reference {
  id: number;
  source_chunk_id: string;
  source_span_start: number | null;
  source_span_end: number | null;
  source_form: string | null;
  quoted_phrase: string | null;
  target: ReferenceTarget;
  reference_type: ReferenceType;
  confidence: number;
  source_method: string;
  curator_status: "auto" | "approved" | "rejected" | "manual";
  note: string | null;
}
 
/** Sparse map shape returned by GET /api/works/:id/references_map. */
export interface ReferencesMap {
  work_id: number;
  lang: string;
  from: number;
  limit: number;
  by_chunk: Record<string, { count: number; top: Reference[] }>;
}
