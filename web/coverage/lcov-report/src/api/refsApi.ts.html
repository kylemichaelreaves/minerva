/** Fetch the curator review queue. Lowest-confidence auto rows first
 *  — those are the ones whose disposition most needs human judgment. */
export async function fetchReviewQueue(
  args: {
    limit?: number;
    offset?: number;
    sourceAuthor?: string;
    referenceType?: string;
    minConfidence?: number;
    maxConfidence?: number;
  } = {},
  signal?: AbortSignal,
): Promise<ReviewQueueResponse> {
  const params = new URLSearchParams();
  if (args.limit !== undefined) params.set("limit", String(args.limit));
  if (args.offset !== undefined) params.set("offset", String(args.offset));
  if (args.sourceAuthor) params.set("source_author", args.sourceAuthor);
  if (args.referenceType) params.set("reference_type", args.referenceType);
  if (args.minConfidence !== undefined)
    params.set("min_confidence", String(args.minConfidence));
  if (args.maxConfidence !== undefined)
    params.set("max_confidence", String(args.maxConfidence));
  const qs = params.toString();
  const url = `/api/references/review_queue${qs ? `?${qs}` : ""}`;
  const r = await fetch(url, { signal });
  if (!r.ok) throw new Error(`failed to fetch review queue: ${r.status}`);
  return (await r.json()) as ReviewQueueResponse;
}
