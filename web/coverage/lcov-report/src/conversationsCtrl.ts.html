// Chat conversation controller. Owns:
//   * messages / history / thinking / pinned signals,
//   * the in-flight `AbortController`,
//   * send / hydrate / delete flows + their server calls,
//   * delete-dialog target state.
//
// Bridges to the router via `route` + `navigate`: hydrates on route
// change to a conversation, navigates to the landing after delete, etc.
//
// `useConversations` must be called from inside a Solid reactive root.
 
import { createEffect, createSignal } from "solid-js";
import {
  chat,
  deleteConversation,
  errorTurn,
  getConversation,
  listConversations,
  nowTime,
} from "./api/chatApi";
import { apiError } from "./lib/apiError";
import { renumberCitations } from "./lib/renumber";
import { isAbortError } from "./util";
import type { Route, RoutingApi } from "./routing";
import type {
  ChatHistoryMessage,
  Conversation,
  HistoryItem,
  PersistedMessage,
  PoietesTurn,
  Turn,
  UserTurn,
} from "./lib/types";
 
function relativeTime(epochSeconds: number): string {
  const now = Date.now() / 1000;
  const delta = now - epochSeconds;
  if (delta < 60) return "Just now";
  if (delta < 60 * 60) return `${Math.floor(delta / 60)}m ago`;
  if (delta < 60 * 60 * 24) return `${Math.floor(delta / 3600)}h ago`;
  if (delta < 60 * 60 * 24 * 2) return "Yesterday";
  if (delta < 60 * 60 * 24 * 7) return `${Math.floor(delta / 86400)}d ago`;
  const d = new Date(epochSeconds * 1000);
  return d.toLocaleDateString(undefined, { month: "short", day: "numeric" });
}
 
function conversationToHistoryItem(c: Conversation): HistoryItem {
  return {
    id: c.id,
    title: c.title,
    meta: relativeTime(c.updated_at),
    dominant: c.dominant ?? "P",
  };
}
 
// Convert a persisted message (server shape) into the UI's Turn shape.
function persistedToTurn(m: PersistedMessage): Turn {
  const time = new Date(m.created_at * 1000).toLocaleTimeString([], {
    hour: "2-digit",
    minute: "2-digit",
  });
  if (m.role === "user") {
    return { role: "user", text: m.content, time };
  }
  // Apply the same renumbering the live chat path does so hydrated
  // threads number their footnotes by order of first mention.
  const renumbered = renumberCitations({
    paragraphs: m.content
      .split("\n\n")
      .map((p) => p.trim())
      .filter(Boolean),
    citations: m.citations,
  });
  return {
    role: "poietes",
    time,
    dominant: m.dominant ?? "P",
    paragraphs: renumbered.paragraphs,
    citations: renumbered.citations,
    abstained: m.abstained ?? false,
  };
}
 
export interface ConversationsApi {
  messages: () => Turn[];
  history: () => HistoryItem[];
  thinking: () => boolean;
  pinnedChunkIds: () => string[];
  deleteTarget: () => HistoryItem | null;
  /** Conversation id from the current route, or `null` outside chat. */
  activeId: () => string | null;
  /** True when on the landing page (no active conversation in the URL). */
  showLanding: () => boolean;
  /** True while a conversation's messages are being fetched from the server. */
  hydrating: () => boolean;
  togglePin: (chunkId: string) => void;
  newConversation: () => void;
  pickHistory: (id: string) => void;
  startFromLanding: (text: string) => void;
  sendMessage: (text: string) => void;
  requestDelete: (item: HistoryItem) => void;
  cancelDelete: () => void;
  confirmDelete: () => void;
  /** Abort any in-flight chat request. Called by the reader controller
   *  when switching modes mid-flight. */
  abortInflight: () => void;
}
 
export function useConversations(routing: RoutingApi): ConversationsApi {
  const [messages, setMessages] = createSignal<Turn[]>([]);
  const [history, setHistory] = createSignal<HistoryItem[]>([]);
  const [thinking, setThinking] = createSignal(false);
  const [pinnedChunkIds, setPinnedChunkIds] = createSignal<string[]>([]);
  const [deleteTarget, setDeleteTarget] = createSignal<HistoryItem | null>(null);
 
  // In-flight chat request; aborted whenever the conversation context
  // changes (new conversation, history switch, or another send) so the
  // user doesn't see a stale reply land in the wrong thread.
  let inflightController: AbortController | undefined;
 
  function abortInflight() {
    inflightController?.abort();
    inflightController = undefined;
  }
 
  const activeId = (): string | null => {
    const r = routing.route();
    return r.kind === "conversation" ? r.id : null;
  };
 
  // Stay on the landing page only until the user has actually started
  // composing. Once `messages` has content, transition to the Thread
  // view immediately so the thinking indicator can render — the server
  // takes a beat to mint a conversation_id and navigate the route.
  const showLanding = (): boolean => routing.route().kind === "landing" && messages().length === 0;
 
  // True between navigating to a conversation route and the hydrate()
  // call landing its messages. Used to show skeleton text blocks so
  // the Thread isn't empty during the round-trip.
  const [hydrating, setHydrating] = createSignal(false);
 
  function togglePin(chunkId: string) {
    setPinnedChunkIds((ids) =>
      ids.includes(chunkId) ? ids.filter((x) => x !== chunkId) : [...ids, chunkId],
    );
  }
 
  // Drive hydration off the route. When the URL identifies a conversation
  // and we haven't loaded its messages yet, fetch them. When the URL
  // moves away, clear local state and any in-flight request — including
  // any hydrate() fetch that might still be in flight.
  let lastHydratedId: string | null = null;
  // Separate AbortController for the hydrate() fetch so it can be
  // cancelled independently of askPoietes's in-flight chat request.
  let hydrateController: AbortController | undefined;
  function abortHydrate(): void {
    hydrateController?.abort();
    hydrateController = undefined;
  }
 
  createEffect(() => {
    const r = routing.route();
    if (r.kind === "conversation") {
      if (lastHydratedId !== r.id) {
        lastHydratedId = r.id;
        abortInflight();
        abortHydrate();
        setMessages([]);
        setPinnedChunkIds([]);
        setThinking(false);
        void hydrate(r.id);
      }
    } else {
      // Leaving a conversation — landing or reader. Abort any in-flight
      // hydrate so its fetch is cancelled AND the loading indicator
      // turns off (the `finally` in hydrate() guards on `activeId() ===
      // id`, which is now false; without an explicit reset here the
      // UI stays in the hydrating state until the user navigates back).
      if (lastHydratedId !== null) {
        lastHydratedId = null;
        abortInflight();
        abortHydrate();
        setMessages([]);
        setPinnedChunkIds([]);
        setThinking(false);
        setHydrating(false);
      }
    }
  });
 
  // First-paint bootstrap: load the conversation list.
  void refreshHistory();
 
  async function refreshHistory() {
    try {
      const list = await listConversations();
      setHistory(list.map(conversationToHistoryItem));
    } catch (err) {
      // Non-fatal: the chat path still works without a populated sidebar.
      // Surface a toast so the user knows the sidebar is stale.
      console.warn("failed to load conversations", err);
      apiError("Couldn't load conversation history.");
    }
  }
 
  async function hydrate(id: string) {
    // Each hydrate has its own AbortController so navigating away
    // mid-fetch cancels the round-trip cleanly. The route-change
    // effect calls `abortHydrate()` which aborts this controller.
    const controller = new AbortController();
    hydrateController = controller;
    setHydrating(true);
    try {
      const detail = await getConversation(id, controller.signal);
      // The user may have moved on while this was inflight; only apply
      // if it still matches the active id.
      if (activeId() !== id) return;
      setMessages(detail.messages.map(persistedToTurn));
    } catch (err) {
      if (isAbortError(err)) return;
      console.warn(`failed to hydrate conversation ${id}`, err);
      // Probably 404 — stale link / deleted on another device. Reset.
      // Toast so the user understands why they were bumped to landing.
      apiError("That conversation is no longer available.");
      if (activeId() === id) routing.navigate({ kind: "landing" });
      void refreshHistory();
    } finally {
      // Clear the controller reference if it's still ours (a concurrent
      // hydrate would have replaced it).
      if (hydrateController === controller) {
        hydrateController = undefined;
      }
      // Only un-set the hydrating signal if we're still on this id.
      // The route-change effect handles the "navigated away" case
      // explicitly via its own setHydrating(false).
      if (activeId() === id) setHydrating(false);
    }
  }
 
  function newConversation() {
    routing.navigate({ kind: "landing" });
  }
 
  function pickHistory(id: string) {
    routing.navigate({ kind: "conversation", id });
  }
 
  function startFromLanding(text: string) {
    // activeId stays null until the server mints one in askPoietes.
    const turn: UserTurn = { role: "user", text, time: nowTime() };
    setMessages([turn]);
    void askPoietes(text);
  }
 
  function sendMessage(text: string) {
    const turn: UserTurn = { role: "user", text, time: nowTime() };
    setMessages((m) => [...m, turn]);
    void askPoietes(text);
  }
 
  function requestDelete(item: HistoryItem) {
    setDeleteTarget(item);
  }
 
  function cancelDelete() {
    setDeleteTarget(null);
  }
 
  async function confirmDelete() {
    const target = deleteTarget();
    if (!target) return;
    setDeleteTarget(null);
    const wasActive = activeId() === target.id;
    if (wasActive) {
      // Don't let an in-flight reply against a now-deleted conversation
      // land in the UI.
      abortInflight();
    }
    // Optimistically drop from the sidebar so the click feels responsive.
    // Remember the original index so we can re-insert on API failure
    // (otherwise an optimistic UI silently lies).
    const original = history();
    const originalIndex = original.findIndex((h) => h.id === target.id);
    setHistory((hs) => hs.filter((h) => h.id !== target.id));
    // Clear the "return to last conversation" key if it pointed at the
    // one we just deleted — otherwise Reader → Chat would restore a
    // dangling id. (`readerCtrl` uses this for its switchMode hop.)
    if (localStorage.getItem("poietes:last-conversation") === target.id) {
      localStorage.removeItem("poietes:last-conversation");
    }
    if (wasActive) {
      routing.navigate({ kind: "landing" });
    }
    try {
      await deleteConversation(target.id);
    } catch (err) {
      console.warn(`failed to delete conversation ${target.id}`, err);
      apiError("Couldn't delete that conversation.");
      // Roll back the optimistic removal so the sidebar matches reality.
      if (originalIndex >= 0) {
        setHistory((hs) => {
          if (hs.some((h) => h.id === target.id)) return hs; // already restored
          const next = [...hs];
          next.splice(Math.min(originalIndex, next.length), 0, target);
          return next;
        });
      }
      // Resync from the server in case our local state is stale for
      // other reasons too.
      void refreshHistory();
      return;
    }
    void refreshHistory();
  }
 
  async function askPoietes(text: string) {
    // Snapshot the conversation as the LLM should see it: every turn
    // so far, with the new user turn appended (already added by the
    // caller).
    const history: ChatHistoryMessage[] = messages()
      .map(
        (m): ChatHistoryMessage =>
          m.role === "user"
            ? { role: "user", content: m.text }
            : { role: "assistant", content: m.paragraphs.join("\n\n") },
      )
      .filter((m) => m.content);
    // The latest turn is the current question; the server expects it as
    // `query`, so drop it from the history we send.
    const priorHistory = history.slice(0, -1);
    // Cancel any prior in-flight call before starting this one.
    abortInflight();
    const controller = new AbortController();
    inflightController = controller;
    setThinking(true);
    try {
      const reply: PoietesTurn = await chat({
        query: text,
        history: priorHistory,
        pinnedChunkIds: pinnedChunkIds().length ? pinnedChunkIds() : undefined,
        conversationId: activeId() ?? undefined,
        signal: controller.signal,
      });
      // Only append if this controller is still the active one — otherwise
      // a slow reply that lost the abort race would land in a thread the
      // user has already moved on from.
      if (inflightController === controller) {
        setMessages((m) => [...m, reply]);
        // First turn: server minted a conversation_id; promote it via
        // the router so the URL updates and hydration tracking stays
        // consistent. `lastHydratedId` is updated by the route effect.
        if (reply.conversation_id && !activeId()) {
          lastHydratedId = reply.conversation_id;
          routing.navigate({ kind: "conversation", id: reply.conversation_id });
        }
        // Drop pin IDs the server couldn't resolve — typically stale
        // after a re-ingest. The user can re-pin from the new chunks.
        if (reply.missing_pinned_ids?.length) {
          const stale = new Set(reply.missing_pinned_ids);
          setPinnedChunkIds((ids) => ids.filter((id) => !stale.has(id)));
        }
        // Refresh so the sidebar reflects the new/updated conversation
        // (creation on first turn, updated_at bump on subsequent turns).
        void refreshHistory();
      }
    } catch (err) {
      if (isAbortError(err)) {
        // User cancelled (new conversation / history switch / new send);
        // drop the result silently.
        return;
      }
      // Anything else is unexpected (`chat()` normally returns an error
      // turn for network/server failures rather than throwing). Surface
      // it in the thread instead of rethrowing — an uncaught rejection
      // here would leave the user staring at a frozen "thinking" state
      // with no feedback.
      const msg = err instanceof Error ? err.message : String(err);
      console.error("askPoietes: unexpected error", err);
      if (inflightController === controller) {
        setMessages((m) => [...m, errorTurn(`Unexpected error: ${msg}`)]);
      }
    } finally {
      if (inflightController === controller) {
        inflightController = undefined;
        setThinking(false);
      }
    }
  }
 
  return {
    messages,
    history,
    thinking,
    pinnedChunkIds,
    deleteTarget,
    activeId,
    showLanding,
    hydrating,
    togglePin,
    newConversation,
    pickHistory,
    startFromLanding,
    sendMessage,
    requestDelete,
    cancelDelete,
    confirmDelete: () => void confirmDelete(),
    abortInflight,
  };
}
