// App composition root. Wires four composables — theme, routing,
// conversations controller, reader controller — into the layout tree.
// State logic lives in the composables; this file is mostly JSX plus
// a handful of UI-only signals (sidebar collapsed, scroll position).
 
import { createEffect, createMemo, createSignal, Show } from "solid-js";
import Sidebar from "./Sidebar";
import Topbar from "./Topbar";
import Landing from "./Landing";
import Composer from "./Composer";
import Thread from "./Thread";
import ConfirmDialog from "./ConfirmDialog";
import Reader from "./Reader";
import Refs from "./Refs";
import Review from "./Refs/Review";
import Toasts from "./lib/Toasts";
import { useTheme } from "./theme";
import { useRouting } from "./routing";
import { useConversations } from "./conversationsCtrl";
import { useReader } from "./readerCtrl";
import type { AccentKey } from "./lib/types";
 
const ACCENT_KEY: AccentKey = "oxblood";
const MARGIN_NOTES = true;
 
export default function App() {
  const themeApi = useTheme(ACCENT_KEY);
  const routing = useRouting();
  const conversations = useConversations(routing);
  const reader = useReader(routing, conversations.abortInflight);
 
  // UI-only state: chrome that doesn't belong to any persistent
  // controller and never needs to outlive the App component.
  const [sidebarCollapsed, setSidebarCollapsed] = createSignal(false);
  const [scrolled, setScrolled] = createSignal(false);
 
  let scrollEl: HTMLDivElement | undefined;
 
  // Auto-scroll to the bottom on new turns.
  createEffect(() => {
    conversations.messages();
    conversations.thinking();
    if (scrollEl) {
      // queueMicrotask so the new DOM nodes are in place before we scroll.
      const el = scrollEl;
      queueMicrotask(() => {
        el.scrollTop = el.scrollHeight;
      });
    }
  });
 
  function onScroll(e: Event & { currentTarget: HTMLDivElement }) {
    setScrolled(e.currentTarget.scrollTop > 4);
  }
 
  const conversationTitle = createMemo<string | null>(() => {
    const id = conversations.activeId();
    if (!id) return null;
    const h = conversations.history().find((x) => x.id === id);
    return h ? h.title : null;
  });
 
  // The cross-references browser is a sibling of reader/chat, dispatched
  // directly off the route rather than through readerCtrl's mode signal.
  // This keeps AppMode narrow (chat | reader) — the /refs view is a
  // standalone page, not a third mode anyone toggles between.
  const onRefs = createMemo(() => {
    const k = routing.route().kind;
    return k === "refs" || k === "refs-review";
  });
 
  return (
    <div class="app">
      <Sidebar
        collapsed={sidebarCollapsed()}
        onToggle={() => setSidebarCollapsed((c) => !c)}
        history={conversations.history()}
        activeId={conversations.activeId()}
        onPick={conversations.pickHistory}
        onNew={conversations.newConversation}
        onRequestDelete={conversations.requestDelete}
        mode={reader.mode()}
        onMode={(m) => void reader.switchMode(m)}
        works={reader.works()}
        activeWorkId={reader.readerWork()?.workId ?? null}
        activeWorkLang={reader.readerWork()?.sourceLang}
        onPickWork={reader.openWork}
        onOpenRefs={() => { routing.navigate({ kind: "refs" }); }}
        onRefs={onRefs()}
      />
      <main class="main">
        <Show
          when={onRefs()}
          fallback={
            <Show
              when={reader.mode() === "reader" && reader.readerWork()}
              fallback={
            <>
              <Topbar
                scrolled={scrolled()}
                sidebarCollapsed={sidebarCollapsed()}
                onToggleSidebar={() => setSidebarCollapsed(false)}
                theme={themeApi.theme()}
                onToggleTheme={themeApi.toggleTheme}
                conversationTitle={!conversations.showLanding() && conversationTitle()}
              />
              <Show
                when={conversations.showLanding()}
                fallback={
                  <>
                    <div class="scroll" ref={scrollEl} onScroll={onScroll}>
                      <Thread
                        messages={conversations.messages()}
                        thinking={conversations.thinking()}
                        hydrating={conversations.hydrating()}
                        marginNotes={MARGIN_NOTES}
                        pinnedChunkIds={conversations.pinnedChunkIds()}
                        onTogglePin={conversations.togglePin}
                        onOpenReader={(workId, sourceLang, chunkId) => {
                          reader.openAtChunk(workId, sourceLang, chunkId);
                        }}
                      />
                    </div>
                    {/* Composer stays enabled while a request is in flight: a new
                        send is the canonical way to replace the slow one
                        (`askPoietes` calls `abortInflight()` before each turn).
                        Visual feedback comes from the thinking-dots indicator
                        inside Thread, not from disabling input. */}
                    <Composer onSend={conversations.sendMessage} />
                  </>
                }
              >
                <Landing
                  onSubmit={conversations.startFromLanding}
                  onPickPrompt={conversations.startFromLanding}
                />
              </Show>
            </>
          }
            >
              <Reader
                workId={reader.readerWork()!.workId}
                sourceLang={reader.readerWork()!.sourceLang}
                focusChunkId={reader.readerWork()!.focusChunkId}
                openTermId={reader.readerWork()!.openTermId}
                onOpenTermIdChange={reader.setOpenTermId}
                onClose={() => void reader.switchMode("chat")}
              />
            </Show>
          }
        >
          {/* Refs routes. Two sibling pages share this slot — the
              corpus browser (/refs) and the curator review queue
              (/refs/review). Dispatched by route kind. */}
          {(() => {
            const r = routing.route();
            if (r.kind === "refs") {
              return (
                <Refs
                  sourceAuthor={r.sourceAuthor}
                  targetAuthor={r.targetAuthor}
                  targetWork={r.targetWork}
                  referenceType={r.referenceType}
                  includeRejected={r.includeRejected}
                  onNavigate={(patch) => {
                    routing.navigate({ kind: "refs", ...patch });
                  }}
                  onOpenReview={() => {
                    routing.navigate({ kind: "refs-review" });
                  }}
                  onOpenSource={(workTitle, chunkId) => {
                    reader.openAtChunk(workTitle, undefined, chunkId);
                  }}
                />
              );
            }
            if (r.kind === "refs-review") {
              return (
                <Review
                  sourceAuthor={r.sourceAuthor}
                  referenceType={r.referenceType}
                  onNavigate={(patch) => {
                    routing.navigate({ kind: "refs-review", ...patch });
                  }}
                  onOpenSource={(workTitle, chunkId) => {
                    reader.openAtChunk(workTitle, undefined, chunkId);
                  }}
                />
              );
            }
            return null;
          })()}
        </Show>
      </main>
 
      <ConfirmDialog
        open={conversations.deleteTarget() !== null}
        title="Delete this conversation?"
        message="This can't be undone."
        confirmLabel="Delete"
        onConfirm={conversations.confirmDelete}
        onCancel={conversations.cancelDelete}
      />
 
      <Toasts />
    </div>
  );
}
