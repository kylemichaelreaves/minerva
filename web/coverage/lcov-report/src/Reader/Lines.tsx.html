// Reader stage: per-chunk lines + per-word tokenisation. Each Greek
// (or any non-Latin source) chunk becomes a `.line` whose words are
// individually wrapped — clickable term marks for curated lemmata,
// clickable `.gw` words for everything else. Romanization on hover
// is Greek-only; other source languages get the underline affordance
// without a tooltip.
 
import { For, Show } from "solid-js";
import { HoverTip } from "../lib/HoverTip";
import { romanize, tokenizeWords } from "../lib/greek";
import type {
  LemmaHighlights,
  OccurrenceMap,
  SourceChunk,
  TermOccurrenceSpan,
} from "../lib/types";
import { XrefMarker } from "../lib/XrefMarker";
import { classnames } from "../util";
import { occurrenceFor } from "./utils";
 
/** Per-chunk reference summary fed in from the references_map fetch.
 *  Lines.tsx only needs the count for rendering the marker; the full
 *  payload (target metadata, quoted phrase, etc.) is fetched by the
 *  caller on click and passed into a ReferenceCard. */
export interface XrefSummary {
  count: number;
}
export type XrefMap = Record<string, XrefSummary>;
 
export function Lines(props: {
  chunks: SourceChunk[];
  lang: string;
  occMap: OccurrenceMap | null;
  /** When non-null, the reader paints a highlight on every word
   *  whose offset matches a span in `by_chunk[chunkId]`. The lemma
   *  itself is in `lemmaHighlights.lemma` for css/aria use. */
  lemmaHighlights: LemmaHighlights | null;
  /** Selection-driven pinned highlights painted by the Reader after
   *  a translation commits. One entry per chunk the selection
   *  touched. `saved=true` means the rail has been dismissed; the
   *  marker drops to a hairline underline. */
  pinnedHighlights?: import("./utils").PinnedHighlight[];
  /** Click handler for a saved-underline marker — re-opens the
   *  rail with the cached translation. */
  onPinnedClick?: (chunkId: string, start: number, end: number) => void;
  /** Optional map of chunk_id → outgoing-reference summary. When
   *  omitted (the common path until the Reader wires
   *  `/api/works/:id/references_map` in), no markers render and the
   *  24px gutter column stays empty — the page never shifts. */
  xrefMap?: XrefMap;
  /** chunk_id whose ReferenceCard is currently open. The marker on
   *  that row renders in active state (accent) while the card is
   *  anchored. */
  openRefChunkId?: string | null;
  /** Click handler for a marker. Receives the chunk_id and the
   *  marker's bounding rect so the caller can fetch the references
   *  for that chunk and anchor the ReferenceCard popover. */
  onXrefOpen?: (chunkId: string, anchor: DOMRect) => void;
  onTermClick: (termId: number, anchor: DOMRect) => void;
  onWordClick: (
    word: string,
    chunkId: string,
    offset: number,
    lineEl: Element | null,
  ) => void;
}) {
  const isGreek = props.lang === "grc";
  const isLatin = !isGreek;
  return (
    <div class={classnames("lines", isLatin && "latin")}>
      <For each={props.chunks}>
        {(chunk) => {
          const xref = (): XrefSummary | undefined =>
            props.xrefMap?.[chunk.chunk_id];
          return (
            <div class="line" data-chunk-id={chunk.chunk_id}>
              <span class="num">{chunk.section ?? chunk.ord + 1}</span>
              <span class="text">
                <WordRun
                  text={chunk.text}
                  lang={props.lang}
                  chunkId={chunk.chunk_id}
                  occurrences={props.occMap?.by_chunk[chunk.chunk_id] ?? []}
                  highlights={props.lemmaHighlights?.by_chunk[chunk.chunk_id] ?? []}
                  pinned={(props.pinnedHighlights ?? []).filter(
                    (p) => p.chunkId === chunk.chunk_id,
                  )}
                  onPinnedClick={props.onPinnedClick}
                  onTermClick={props.onTermClick}
                  onWordClick={props.onWordClick}
                />
              </span>
              {/* The 24px right-gutter cell is always present (grid
                  reserves the column) so the page never shifts as
                  markers come and go. The marker itself only renders
                  when this chunk has outgoing references AND the caller
                  has wired an open-handler. */}
              <span class="line-marker">
                <Show when={xref() && props.onXrefOpen}>
                  <XrefMarker
                    count={xref()!.count}
                    active={props.openRefChunkId === chunk.chunk_id}
                    onOpen={(rect) => {
                      props.onXrefOpen!(chunk.chunk_id, rect);
                    }}
                  />
                </Show>
              </span>
            </div>
          );
        }}
      </For>
    </div>
  );
}
 
// Splits a chunk into word / non-word runs. Each word either:
//   - falls inside a term-occurrence span → wrapped in a clickable
//     .term-mark with no hover tooltip (the click is the gesture);
//   - or is an ordinary word → wrapped in `<span class="gw">` for the
//     dashed-underline hover affordance. Greek gets a HoverTip on top
//     carrying its romanization; other source languages (German,
//     etc.) skip the tooltip — the underline alone confirms the word
//     is interactive while a term card is open, without invading the
//     reading flow.
// Selection still works across all spans, so the mouseUp → translate
E// flow is unaffected.

   *  surface_start/surface_end exactly), so an inclusive containment
   *  check is enough. */

  pinned: import("./utils").PinnedHighlight[];

      if (h.start <= offset && offset + length <= h.end) return true;

        // calls inside JSX, not bare variable refs at row-init time.

          const lemma = isHighlighted(t.offset, t.s.length) ? " lemma-hl" : "";
          const p = pin();

          return `${extra}${lemma}${pCls}`;
        };

          e: MouseEvent & { currentTarget: HTMLSpanElement },
        ) => {

            onPinnedClickInner(e, p);
            return;

          onClickWord(e);
        };
        return (
          <Show
            when={occurrenceFor(props.occurrences, t.offset, t.s.length)}
            fallback={
              isGreek ? (
                <HoverTip body={romanize(t.s)}>
                  <span
                    class={baseClass("gw")}
                    data-offset={t.offset}
                    onClick={handleClick}
                  >
                    {t.s}
                  </span>
